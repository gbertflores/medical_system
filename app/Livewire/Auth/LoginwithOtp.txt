<?php

namespace App\Livewire\Auth;

use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Mail;
use Illuminate\Validation\ValidationException;
use Livewire\Component;

class Login extends Component
{

    public $email;
    public $password;
    public $otp;
    public $isOtpSent = false;
    public $user;

    protected $rules= [
        'email' => 'required|email',
        'password' => 'required'

    ];

    public function render()
    {
        return view('livewire.auth.login');
    }

    // public function mount() {
      
    //     $this->fill(['email' => 'admin@material.com', 'password' => 'secret']);    
    // }
    
    public function store()
    {
        $attributes = $this->validate();

        if (! auth()->attempt($attributes)) {
            throw ValidationException::withMessages([
                'email' => 'Your provided credentials could not be verified.'
            ]);
        }
        $this->user = Auth::user(); //new
        $this->sendOtp(); // new

        // session()->regenerate();
        
        // if (is_null(auth()->user()->username))
        //     return redirect('/setup-account');

        // return redirect('/dashboard');

    }

    public function sendOtp()
    {
        $this->otp = rand(100000, 999999);
        $this->user->update(['otp' => $this->otp]);

        // Send OTP email
        Mail::to($this->user->email)->send(new \App\Mail\SendOtpMail($this->otp));

        session()->flash('status', 'OTP sent to your email.');

        $this->isOtpSent = true;
    }

    public function verifyOtp()
    {
        $this->validate([
            'otp' => 'required|digits:6',
        ]);

        if ($this->otp == $this->user->otp) {
            $this->user->update(['otp' => null]);
            session()->regenerate();

            if (is_null(auth()->user()->username))
                return redirect('/setup-account');

            session()->flash('status', 'Logged in successfully.');
            return redirect()->route('dashboard');
        }

        throw ValidationException::withMessages([
            'otp' => 'Invalid OTP. Please try again.',
        ]);
    }
}
